\newcommand\WornItemsTableAdd[3]{%
  \g@addto@macro\WornItemsTable{\footnotesize #1 & \footnotesize #2 & \footnotesize #3 \\ \hline }
}
\makeatother

\newcommand{\AddWornItem}[3]{
    \WornItemsTableAdd{#1}{#2}{#3}
    \ifblank{#2}{}{\stepcounter{Invest}}
    \ifstrempty{#3}{}{\ifstrequal{#3}{L}{}{\FPadd\bulk#3\bulk}}
}
